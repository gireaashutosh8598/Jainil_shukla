function thinkingModule(T, key) {
  const qa = T.questions.map(([q, why, led, tg], i) => `<details class="tq"${i === 0 ? ' open' : ''}><summary><span class="tq-q">Q${i + 1}</span><span class="tq-t">${esc(q)}</span></summary><div class="tq-b"><div><span class="t-label">Why it mattered</span><p>${why}</p></div><div><span class="t-label">What it led to</span><p>${led}</p>${tag(tg)}</div></div></details>`).join('');
  const cards = (arr, a, b) => `<div class="grid g-3 tk-grid">${arr.map(([t, x, y, tg]) => `<div class="card tk-card"><div class="row between" style="gap:8px;align-items:flex-start"><h4 class="t-h4">${esc(t)}</h4>${tag(tg)}</div><span class="t-label">${a}</span><p class="t-sm">${x}</p><span class="t-label">${b}</span><p class="t-sm ink2">${y}</p></div>`).join('')}</div>`;
  const panels = [
    { t: 'Working pattern', html: `<p class="tk-intro">The sequence I followed on this project.</p><ol class="pattern">${T.pattern.map(([a, b], i) => `<li><span class="pn">${String(i + 1).padStart(2, '0')}</span><strong>${esc(a)}</strong><span>${esc(b)}</span></li>`).join('')}</ol>` },
    { t: 'Critical thinking', html: `<p class="tk-intro">The questions that shaped the work, and where each one led.</p>${qa}` },
    { t: 'Research techniques', html: `<p class="tk-intro">How I gathered evidence, and what each method revealed.</p>${cards(T.research, 'How I used it', 'What it revealed')}` },
    { t: 'Consumer behaviour', html: `<p class="tk-intro">What I paid attention to about the people at the end of the system.</p>${cards(T.consumer, 'Observation', 'Implication')}` },
    { t: 'Insights', html: `<div class="ins-grid">${T.insights.map(([i, e, s], k) => `<article class="ins"><span class="ins-n">${String(k + 1).padStart(2, '0')}</span><p class="ins-t">${esc(i)}</p><dl><dt>Evidence</dt><dd>${esc(e)}</dd><dt>So what</dt><dd>${esc(s)}</dd></dl></article>`).join('')}</div>` },
    { t: 'Strategy planning', html: `<p class="tk-intro">How the thinking was sequenced into a plan. Status shows what happened and what is still proposed.</p><ol class="road">${T.strategy.map(([w, t, d, tg]) => `<li class="road-${tg}"><span class="rw t-label">${esc(w)}</span><span class="rd" aria-hidden="true"></span><div><strong>${esc(t)}</strong><p class="t-sm ink2">${esc(d)}</p>${tag(tg)}</div></li>`).join('')}</ol>` },
    { t: 'Solutions', html: `<div class="sol"><div class="sol-h t-label"><span>Problem</span><span></span><span>Solution</span><span>Status</span></div>${T.solutions.map(([p, s, tg]) => `<div class="sol-r"><span>${esc(p)}</span><span class="sol-a" aria-hidden="true">→</span><strong>${esc(s)}</strong><span>${tag(tg)}</span></div>`).join('')}</div>` }
  ];
  return `<div class="standout"><div><span class="t-label">How I stood out · ${esc(T.lens)}</span><p class="standout-t">${esc(T.standout)}</p></div><div class="standout-pts">${T.points.map(([a, b], i) => `<div><span class="t-label" style="color:var(--blue-ink)">0${i + 1}</span><h4 class="t-h4">${esc(a)}</h4><p class="t-sm ink2">${esc(b)}</p></div>`).join('')}</div></div>${tabs(panels, 'How I thought')}`;
}

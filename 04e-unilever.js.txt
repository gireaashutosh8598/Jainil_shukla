function wSwimlane(el, cfg) {
  const n = cfg.cols.length; let sel = null;
  const cell = (r, v) => r.name === 'Status' ? tag(v) : esc(v);
  let grid = `<div class="sl" style="grid-template-columns:150px repeat(${n},minmax(150px,1fr))"><div class="sl-l sl-corner">Lane / Phase</div>${cfg.cols.map((c, k) => `<button class="sl-h" data-col="${k}" aria-pressed="false"><span class="t-label">${esc(cfg.when[k])}</span>${String(k + 1).padStart(2, '0')} ${esc(c)}</button>`).join('')}`;
  cfg.rows.forEach(r => { grid += `<div class="sl-l">${esc(r.name)}</div>${r.cells.map((v, k) => `<div class="sl-c" data-col="${k}">${cell(r, v)}</div>`).join('')}`; });
  grid += '</div>';
  const mob = `<div class="sl-mobile"><div class="chips" style="padding:14px 18px" role="group" aria-label="Phase">${cfg.cols.map((c, k) => `<button class="chip" data-m="${k}" aria-pressed="${k === 0}">${String(k + 1).padStart(2, '0')} ${esc(c)}</button>`).join('')}</div><div class="bp-stack"></div></div>`;
  el.innerHTML = frame(cfg.title, tag('done'), `<div class="sl-wrap">${grid}</div>${mob}`, '<span>Select a phase to follow it down every lane</span><span>Read each column top to bottom: question → method → map → output.</span>');
  const stack = $('.sl-mobile .bp-stack', el);
  const pm = k => { stack.innerHTML = `<div class="fp-anim"><span class="t-label">${esc(cfg.when[k])}</span><strong>${esc(cfg.cols[k])}</strong></div>` + cfg.rows.map(r => `<div class="fp-anim"><span class="t-label">${esc(r.name)}</span>${cell(r, r.cells[k])}</div>`).join(''); $$('[data-m]', el).forEach(b => b.setAttribute('aria-pressed', +b.dataset.m === k)); };
  $$('[data-m]', el).forEach(b => b.addEventListener('click', () => pm(+b.dataset.m))); pm(0);
  $$('.sl-h', el).forEach(b => b.addEventListener('click', () => { const k = +b.dataset.col; sel = sel === k ? null : k; $$('.sl-h', el).forEach(h => h.setAttribute('aria-pressed', +h.dataset.col === sel)); $$('.sl-c', el).forEach(c => c.classList.toggle('on', +c.dataset.col === sel)); }));
}
WIDGETS.swimlane = wSwimlane;

function wMoments(el, cfg) {
  const it = cfg.items, n = it.length; let i = 0;
  const X = k => 40 + k * (520 / (n - 1));
  const pts = it.map((m, k) => [X(k), 20 + m.y * 1.4]);
  let d = `M${pts[0][0]},${pts[0][1]}`; for (let k = 1; k < n; k++) { const [x0, y0] = pts[k - 1], [x1, y1] = pts[k]; const mx = (x0 + x1) / 2; d += ` C${mx},${y0} ${mx},${y1} ${x1},${y1}`; }
  el.innerHTML = frame(cfg.title, tag('illus', 'Conceptual, not measured'), `<div class="mo"><svg class="mo-svg" viewBox="0 0 600 190" role="img" aria-label="Friction curve across five consumer moments"><text x="8" y="22" class="mo-ax">Smooth</text><text x="8" y="178" class="mo-ax">Friction</text><path d="M40 96H560" stroke="var(--line-2)" stroke-dasharray="4 4"/><path d="${d}" fill="none" stroke="var(--ink)" stroke-width="2.2"/>${pts.map(([x, y], k) => `<g class="mo-pt" data-k="${k}" tabindex="0" role="button" aria-label="${esc(it[k].t)}"><circle cx="${x}" cy="${y}" r="16" fill="transparent"/><circle class="mo-dot ${it[k].s}" cx="${x}" cy="${y}" r="8"/><text x="${x}" y="${y > 100 ? y - 16 : y + 26}" text-anchor="middle" class="mo-lb">${esc(it[k].t)}</text></g>`).join('')}</svg>
    <div class="chips chips-row" role="group" aria-label="Moments">${it.map((m, k) => `<button class="chip" data-c="${k}" aria-pressed="${k === 0}">${esc(m.t)}</button>`).join('')}</div><div class="g-panel" aria-live="polite"></div></div>`,
    `<span>Higher on the curve means smoother; lower means more friction. Select a moment.</span><span>${cfg.note}</span>`);
  const panel = $('.g-panel', el);
  const r = () => { const m = it[i]; $$('.mo-pt', el).forEach((g, k) => g.classList.toggle('on', k === i)); $$('[data-c]', el).forEach((b, k) => b.setAttribute('aria-pressed', k === i)); panel.innerHTML = `<div class="fp-anim"><div class="row between"><h4>${String(i + 1).padStart(2, '0')} · ${esc(m.t)}</h4>${tag(m.s)}</div><p style="margin-top:6px"><strong>Moment:</strong> ${esc(m.d)}</p><p style="margin-top:6px"><strong>Insight:</strong> ${esc(m.i)}</p></div>`; };
  $$('.mo-pt', el).forEach(g => { g.addEventListener('click', () => { i = +g.dataset.k; r(); }); g.addEventListener('keydown', e => { if (e.key === 'Enter' || e.key === ' ') { e.preventDefault(); i = +g.dataset.k; r(); } }); });
  $$('[data-c]', el).forEach(b => b.addEventListener('click', () => { i = +b.dataset.c; r(); }));
  r();
}

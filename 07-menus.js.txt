    <aside class="drawer drawer-wide" id="drawer-work" role="dialog" aria-modal="true" aria-labelledby="dt-work" hidden>
      <header class="drawer-h"><div><span class="t-label">See the work</span><h2 id="dt-work">Three projects. One way of thinking.</h2></div><button class="drawer-x" type="button" data-drawer-close aria-label="Close">×</button></header>
      <div class="drawer-b">${workMenuHTML()}<a class="btn btn-ink" href="#/work" style="align-self:flex-start">Open the full Work page →</a></div>
    </aside>`;
  document.body.appendChild(wrap);
  mountWidgets($('#drawer-story', wrap));
  const bg = $('.drawer-bg', wrap);
  let open = null, opener = null;
  function show(name, from) {
    hide(true);
    const d = $('#drawer-' + name, wrap); if (!d) return;
    opener = from || document.activeElement; open = d;
    bg.hidden = false; d.hidden = false; document.body.style.overflow = 'hidden';
    requestAnimationFrame(() => { bg.classList.add('on'); d.classList.add('on'); });
    setTimeout(() => { const f = $('.drawer-x', d); if (f) f.focus(); }, 60);
  }
  function hide(silent) {
    if (!open) return;
    const d = open; open = null;
    d.classList.remove('on'); bg.classList.remove('on');
    const done = () => { d.hidden = true; bg.hidden = true; };
    if (RM.matches || silent) done(); else setTimeout(done, 280);
    if (!mnav.classList.contains('open')) document.body.style.overflow = '';
    if (!silent && opener && opener.focus) opener.focus({ preventScroll: true });
  }
  window.openDrawer = show; window.closeDrawer = hide;
  document.addEventListener('click', e => {
    const t = e.target.closest('[data-open]');
    if (t && !e.metaKey && !e.ctrlKey && !e.shiftKey) { e.preventDefault(); closeMenu(); show(t.dataset.open, t); return; }
    if (e.target.closest('[data-drawer-close]')) { hide(); return; }
    if (open && e.target.closest('.drawer a[href^="#"]')) hide(true);
  });
  document.addEventListener('keydown', e => {
    if (!open) return;
    if (e.key === 'Escape') { hide(); return; }
    if (e.key === 'Tab') {
      const f = $$('a[href],button:not([disabled]),summary', open).filter(x => x.offsetParent !== null);
      if (!f.length) return; const first = f[0], last = f[f.length - 1];
      if (e.shiftKey && document.activeElement === first) { e.preventDefault(); last.focus(); }
      else if (!e.shiftKey && document.activeElement === last) { e.preventDefault(); first.focus(); }
    }
  });
  $$('[data-djump]', wrap).forEach(b => b.addEventListener('click', () => { const t = document.getElementById(b.dataset.djump); const body = t.closest('.drawer-b'); body.scrollTo({ top: t.offsetTop - body.offsetTop - 12, behavior: RM.matches ? 'auto' : 'smooth' }); }));
  $('[data-copy-email]', wrap).addEventListener('click', async () => {
    const m = $('[data-copy-msg]', wrap);
    try { await navigator.clipboard.writeText(EMAIL); m.textContent = 'Copied'; } catch (_) { m.textContent = EMAIL; }
    setTimeout(() => m.textContent = '', 2500);
  });
  $('[data-openbot-menu]', wrap).addEventListener('click', () => { hide(true); const l = document.querySelector('.jv-launch'); if (l && l.getAttribute('aria-expanded') !== 'true') l.click(); });
  window.addEventListener('hashchange', () => hide(true));
})();

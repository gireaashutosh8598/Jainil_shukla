const $ = (s, r = document) => r.querySelector(s);
const $$ = (s, r = document) => Array.from(r.querySelectorAll(s));
const clamp = (v, a, b) => Math.max(a, Math.min(b, v));
const esc = s => String(s ?? '').replace(/[&<>"']/g, c => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]));
const RM = window.matchMedia('(prefers-reduced-motion: reduce)');
const store = {
  get(k, d) { try { const v = localStorage.getItem(k); return v ? JSON.parse(v) : d; } catch (e) { return d; } },
  set(k, v) { try { localStorage.setItem(k, JSON.stringify(v)); } catch (e) {} }
};
const TAGS = {
  done: 'Completed work', research: 'Research & observation', rec: 'Recommendation', concept: 'Conceptual proposal',
  ongoing: 'In development', upcoming: 'Upcoming', ambition: 'Future ambition', illus: 'Illustrative', hyp: 'Unverified hypothesis'
};
const tag = (k, label) => `<span class="tag tag-${k}">${esc(label || TAGS[k])}</span>`;
const ph = t => `<span class="ph">[${esc(t)}]</span>`;
const legend = (keys = Object.keys(TAGS)) => `<div class="legend" role="list" aria-label="Evidence key">${keys.map(k => `<span role="listitem">${tag(k)}</span>`).join('')}</div>`;

function frame(title, tools = '', body = '', hint = '') {
  return `<div class="w-frame"><div class="w-bar"><span class="t-label">${title}</span><div class="w-tools">${tools}</div></div>${body}${hint ? `<div class="w-hint">${hint}</div>` : ''}</div>`;
}

/* =========================================================
   PAN / ZOOM VIEWPORT
   ========================================================= */
function panZoom(vp, stage, opts = {}) {
  let s = 1, x = 0, y = 0, fitS = 1;
  let ptr = null, moved = false, sx = 0, sy = 0, ox = 0, oy = 0;
  function apply(anim) {
    stage.classList.toggle('anim', !!anim && !RM.matches);
    stage.style.transform = `translate(${x}px,${y}px) scale(${s})`;
    vp.classList.toggle('zoomed', s > fitS * 1.04);
  }
  function fit(anim) {
    const r = vp.getBoundingClientRect();
    const W = stage.offsetWidth || 1, H = stage.offsetHeight || 1;
    fitS = Math.min(r.width / W, r.height / H, opts.maxFit ?? 1);
    s = fitS; x = (r.width - W * s) / 2; y = (r.height - H * s) / 2;
    apply(anim);
  }
  function zoomAt(f, cx, cy, anim) {
    const ns = clamp(s * f, fitS * .7, Math.max(2.6, fitS * 4));
    x = cx - (cx - x) * (ns / s); y = cy - (cy - y) * (ns / s); s = ns; apply(anim);
  }
  function zoomBy(f) { const r = vp.getBoundingClientRect(); zoomAt(f, r.width / 2, r.height / 2, true); }
  vp.addEventListener('pointerdown', e => {
    if (e.button !== 0) return;
    if (e.pointerType === 'touch' && s <= fitS * 1.04) return; // keep native page scroll on touch until zoomed
    ptr = e.pointerId; moved = false; sx = e.clientX; sy = e.clientY; ox = x; oy = y;
  });
  vp.addEventListener('pointermove', e => {
    if (ptr !== e.pointerId) return;
    const dx = e.clientX - sx, dy = e.clientY - sy;
    if (!moved && Math.hypot(dx, dy) > 5) { moved = true; vp.classList.add('dragging'); try { vp.setPointerCapture(ptr); } catch (_) {} }
    if (moved) { x = ox + dx; y = oy + dy; apply(false); }
  });
  const end = e => { if (ptr !== e.pointerId) return; ptr = null; vp.classList.remove('dragging'); setTimeout(() => moved = false, 0); };
  vp.addEventListener('pointerup', end); vp.addEventListener('pointercancel', end);
  vp.addEventListener('click', e => { if (moved) { e.stopPropagation(); e.preventDefault(); } }, true);
  vp.addEventListener('wheel', e => {
    if (!(e.ctrlKey || e.metaKey)) return;
    e.preventDefault(); const r = vp.getBoundingClientRect();
    zoomAt(Math.exp(-e.deltaY * .0025), e.clientX - r.left, e.clientY - r.top, false);
  }, { passive: false });
  return { fit, zoomIn: () => zoomBy(1.25), zoomOut: () => zoomBy(.8) };
}
const zoomTools = () => `<button class="btn btn-sm btn-icon" data-z="out" aria-label="Zoom out">−</button><button class="btn btn-sm btn-icon" data-z="in" aria-label="Zoom in">+</button><button class="btn btn-sm" data-z="fit">Fit</button>`;
function bindZoom(root, pz) {
  $$('[data-z]', root).forEach(b => b.addEventListener('click', () => { const z = b.dataset.z; z === 'in' ? pz.zoomIn() : z === 'out' ? pz.zoomOut() : pz.fit(true); }));
}
const ZOOM_HINT = '<span>Drag to pan · Ctrl/⌘ + scroll or the +/− buttons to zoom</span>';

/* =========================================================
   GRAPH — network / ecosystem / causal / stakeholder maps
   ========================================================= */
function wGraph(el, cfg) {
  const id = 'g' + Math.random().toString(36).slice(2, 7);
  const loops = cfg.loops ? cfg.loops.map((l, i) => `<button class="btn btn-sm" data-loop="${i}" aria-pressed="false">${esc(l.name)}</button>`).join('') : '';
  el.innerHTML = frame(cfg.title, loops + zoomTools(),
    `<div class="vp" style="height:var(--vh)"><div class="vp-stage"><svg class="vp-edges" aria-hidden="true"><defs><marker id="${id}a" viewBox="0 0 10 10" refX="9" refY="5" markerWidth="7" markerHeight="7" orient="auto-start-reverse"><path d="M0 0L10 5L0 10z" fill="context-stroke"/></marker></defs></svg></div></div>
     <div class="chips-row"><div class="chips" role="group" aria-label="Nodes">${cfg.nodes.map(n => `<button class="chip" data-chip="${n.id}" aria-pressed="false">${esc(n.label)}</button>`).join('')}</div></div>
     <div class="g-panel" aria-live="polite"></div>`, cfg.hint || ZOOM_HINT);
  const vp = $('.vp', el), stage = $('.vp-stage', el), svg = $('svg', el), panel = $('.g-panel', el);
  const nodeEls = {};
  cfg.nodes.forEach(n => {
    const b = document.createElement('button');
    b.className = 'g-node ' + (n.cls || ''); b.dataset.id = n.id;
    b.innerHTML = `${esc(n.label)}${n.sub ? `<small>${esc(n.sub)}</small>` : ''}`;
    b.setAttribute('aria-label', n.label + (n.sub ? ', ' + n.sub : ''));
    stage.appendChild(b); nodeEls[n.id] = b;
  });
  const edges = cfg.edges.map(e => ({ a: e[0], b: e[1], label: e[2] || '', dash: e[3] }));
  let sel = null, loopSel = null;
  const neighbors = idv => new Set(edges.filter(e => e.a === idv || e.b === idv).map(e => e.a === idv ? e.b : e.a));
  function layout() {
    const r = vp.getBoundingClientRect(); const mob = r.width < 640;
    const H = mob ? (cfg.mh || 560) : (cfg.h || 520);
    vp.style.setProperty('--vh', H + 'px');
    const W = r.width;
    stage.style.width = W + 'px'; stage.style.height = H + 'px';
    svg.setAttribute('width', W); svg.setAttribute('height', H); svg.setAttribute('viewBox', `0 0 ${W} ${H}`);
    const P = {};
    cfg.nodes.forEach(n => {
      const px = (mob && n.m ? n.m[0] : n.x) / 100 * W, py = (mob && n.m ? n.m[1] : n.y) / 100 * H;
      const e2 = nodeEls[n.id]; e2.style.left = px + 'px'; e2.style.top = py + 'px';
      P[n.id] = { x: px, y: py, w: e2.offsetWidth / 2 + 4, h: e2.offsetHeight / 2 + 4 };
    });
    $$('path,text', svg).forEach(p => { if (!p.closest('marker')) p.remove(); });
    edges.forEach((e, i) => {
      const A = P[e.a], B = P[e.b]; if (!A || !B) return;
      const dx = B.x - A.x, dy = B.y - A.y;
      const tA = Math.min(A.w / Math.abs(dx || 1e-6), A.h / Math.abs(dy || 1e-6));
      const tB = Math.min(B.w / Math.abs(dx || 1e-6), B.h / Math.abs(dy || 1e-6));
      const x1 = A.x + dx * tA, y1 = A.y + dy * tA, x2 = B.x - dx * tB, y2 = B.y - dy * tB;
      const curve = cfg.curve ? cfg.curve : 0;
      const mx = (x1 + x2) / 2 - dy * curve, my = (y1 + y2) / 2 + dx * curve;
      const p = document.createElementNS('http://www.w3.org/2000/svg', 'path');
      p.setAttribute('d', curve ? `M${x1},${y1} Q${mx},${my} ${x2},${y2}` : `M${x1},${y1} L${x2},${y2}`);
      p.setAttribute('class', 'g-edge' + (e.dash ? ' dash' : '')); p.dataset.i = i;
      if (cfg.directed) p.setAttribute('marker-end', `url(#${id}a)`);
      svg.appendChild(p);
      if (e.label) {
        const t = document.createElementNS('http://www.w3.org/2000/svg', 'text');
        const lx = curve ? (x1 + 2 * mx + x2) / 4 : (x1 + x2) / 2, ly = curve ? (y1 + 2 * my + y2) / 4 : (y1 + y2) / 2;
        t.setAttribute('x', lx); t.setAttribute('y', ly + 4); t.setAttribute('text-anchor', 'middle'); t.setAttribute('class', 'g-elabel'); t.dataset.i = i; t.textContent = e.label;
        svg.appendChild(t);
      }
    });
    paint(); pz.fit(false);
  }
  function paint() {
    let activeNodes = null, activeEdges = null;
    if (loopSel !== null) {
      const L = cfg.loops[loopSel]; activeNodes = new Set(L.nodes);
      activeEdges = new Set(edges.map((e, i) => (activeNodes.has(e.a) && activeNodes.has(e.b) && (L.edges ? L.edges.includes(i) : true)) ? i : -1).filter(i => i >= 0));
    } else if (sel) {
      activeNodes = neighbors(sel); activeNodes.add(sel);
      activeEdges = new Set(edges.map((e, i) => (e.a === sel || e.b === sel) ? i : -1).filter(i => i >= 0));
    }
    Object.entries(nodeEls).forEach(([k, b]) => {
      b.classList.toggle('active', k === sel || (loopSel !== null && activeNodes.has(k) && false));
      b.classList.toggle('dim', !!activeNodes && !activeNodes.has(k));
      b.setAttribute('aria-pressed', k === sel ? 'true' : 'false');
    });
    $$('[data-chip]', el).forEach(c => c.setAttribute('aria-pressed', c.dataset.chip === sel ? 'true' : 'false'));
    $$('.g-edge,.g-elabel', svg).forEach(p => {
      const i = +p.dataset.i; p.classList.toggle('on', !!activeEdges && activeEdges.has(i)); p.classList.toggle('dim', !!activeEdges && !activeEdges.has(i));
    });
    $$('[data-loop]', el).forEach(b => b.setAttribute('aria-pressed', +b.dataset.loop === loopSel ? 'true' : 'false'));
    // panel
    let html;
    if (loopSel !== null) {
      const L = cfg.loops[loopSel];
      html = `<div class="fp-anim"><span class="t-label">Loop</span><h4>${esc(L.name)}</h4><p>${L.desc}</p></div>`;
    } else if (sel) {
      const n = cfg.nodes.find(q => q.id === sel);
      const nb = [...neighbors(sel)].map(k => cfg.nodes.find(q => q.id === k)).filter(Boolean);
      html = `<div class="fp-anim"><div class="row between"><h4>${esc(n.label)}</h4><button class="btn btn-sm btn-secondary" data-clear>Clear</button></div>
        <p style="margin-top:6px">${n.body || ''}</p>${n.tags ? `<div class="g-meta">${n.tags.map(t => tag(...[].concat(t))).join('')}</div>` : ''}
        ${nb.length ? `<div class="g-meta" style="margin-top:12px"><span class="t-label" style="align-self:center;margin-right:4px">Connects to</span>${nb.map(m => `<button class="chip" data-go="${m.id}">${esc(m.label)}</button>`).join('')}</div>` : ''}</div>`;
    } else {
      html = `<div><span class="t-label">${esc(cfg.panelLabel || 'Explore')}</span><p style="margin-top:8px">${cfg.intro || 'Select any node to see what it does and how it connects.'}</p></div>`;
    }
    panel.innerHTML = html;
    $$('[data-go]', panel).forEach(b => b.addEventListener('click', () => select(b.dataset.go)));
    const c = $('[data-clear]', panel); if (c) c.addEventListener('click', () => select(null));
  }
  function select(k) { loopSel = null; sel = (k === sel) ? null : k; paint(); }
  Object.values(nodeEls).forEach(b => b.addEventListener('click', () => select(b.dataset.id)));
  $$('[data-chip]', el).forEach(c => c.addEventListener('click', () => select(c.dataset.chip)));
  $$('[data-loop]', el).forEach(b => b.addEventListener('click', () => { sel = null; loopSel = loopSel === +b.dataset.loop ? null : +b.dataset.loop; paint(); }));
  const pz = panZoom(vp, stage); bindZoom(el, pz);
  new ResizeObserver(() => layout()).observe(el);
  if (cfg.start) { sel = cfg.start; }
  layout();
}

/* =========================================================
   MIND MAP — expandable, two-sided, with list view
   ========================================================= */
function wMindmap(el, cfg) {
  const root = cfg.data; const all = {};
  (function walk(n, d, p) { n._d = d; n._p = p; all[n.id] = n; (n.children || []).forEach(c => walk(c, d + 1, n)); })(root, 0, null);
  const expanded = new Set([root.id, ...(cfg.open || [])]);
  let mode = null, selId = null;
  el.innerHTML = frame(cfg.title,
    `<button class="btn btn-sm" data-mm="expand">Expand all</button><button class="btn btn-sm" data-mm="collapse">Collapse</button><span style="width:6px"></span><button class="btn btn-sm" data-view="map" aria-pressed="true">Map</button><button class="btn btn-sm" data-view="list" aria-pressed="false">List</button><span class="mm-z" style="display:contents">${zoomTools()}</span>`,
    `<div class="vp" style="height:${cfg.h || 560}px"><div class="vp-stage"><svg class="vp-edges" aria-hidden="true"></svg></div></div><div class="mm-list" hidden style="padding:8px 14px;max-height:620px;overflow:auto"></div><div class="g-panel" aria-live="polite"></div>`,
    cfg.hint || '<span>Click a node to expand it and read the detail · Drag to pan · Ctrl/⌘ + scroll to zoom</span>' + (cfg.legend || ''));
  const vp = $('.vp', el), stage = $('.vp-stage', el), svg = $('svg', el), list = $('.mm-list', el), panel = $('.g-panel', el);
  const pz = panZoom(vp, stage, { maxFit: 1 }); bindZoom(el, pz);
  const COL = 222, ROW = 48, OFF = 128;
  function vis(n) { return expanded.has(n.id) ? (n.children || []) : []; }
  function drawMap(newIds = new Set()) {
    $$('.mm-node', stage).forEach(n => n.remove()); svg.innerHTML = '';
    const kids = root.children || []; const half = Math.ceil(kids.length / 2);
    const sides = [{ dir: 1, nodes: kids.slice(0, half) }, { dir: -1, nodes: kids.slice(half) }];
    let maxDepth = { 1: 0, '-1': 0 };
    sides.forEach(sd => {
      let cur = 0;
      const place = (n, d) => { n._dir = sd.dir; n._dd = d; maxDepth[sd.dir] = Math.max(maxDepth[sd.dir], d); const v = vis(n); if (!v.length) { n._y = cur * ROW; cur++; } else { v.forEach(c => place(c, d + 1)); n._y = (v[0]._y + v[v.length - 1]._y) / 2; } };
      sd.nodes.forEach(n => { place(n, 1); cur += .35; });
      sd.h = Math.max(ROW, (cur - .35) * ROW);
    });
    const H = Math.max(sides[0].h, sides[1].h) + 80;
    const rootX = 40 + (maxDepth[-1] ? (maxDepth[-1] - 1) * COL + OFF + 200 : 110);
    const W = rootX + (maxDepth[1] ? (maxDepth[1] - 1) * COL + OFF + 200 : 110) + 40;
    const rootY = H / 2;
    stage.style.width = W + 'px'; stage.style.height = H + 'px';
    svg.setAttribute('width', W); svg.setAttribute('height', H); svg.setAttribute('viewBox', `0 0 ${W} ${H}`);
    sides.forEach(sd => sd.off = rootY - sd.h / 2 + ROW / 2 - ROW / 2);
    const made = [];
    const mk = (n) => {
      const b = document.createElement('button'); const has = (n.children || []).length;
      b.className = 'mm-node' + (n._d === 0 ? ' root' : n._d === 1 ? ' l1' : '') + (n.focus ? ' focus' : '') + (n._dir === -1 && n._d > 0 ? ' left' : '') + (n.id === selId ? ' sel' : '') + (newIds.has(n.id) ? ' mm-new' : '');
      b.innerHTML = `<span>${esc(n.label)}</span>${has && n._d > 0 ? `<span class="tog" aria-hidden="true">${expanded.has(n.id) ? '−' : '+'}</span>` : ''}`;
      if (has) b.setAttribute('aria-expanded', expanded.has(n.id)); b.dataset.id = n.id;
      stage.appendChild(b); made.push([n, b]);
      return b;
    };
    const rb = mk(root); rb.style.left = rootX + 'px'; rb.style.top = rootY + 'px'; rb.style.transform = 'translate(-50%,-50%)';
    root._ax = { r: rootX + rb.offsetWidth / 2, l: rootX - rb.offsetWidth / 2, y: rootY };
    const placeEls = (n) => {
      vis(n).forEach(c => {
        const b = mk(c); const sd = sides.find(s => s.dir === c._dir);
        const y = sd.off + c._y; let x;
        if (c._dir === 1) { x = rootX + OFF + (c._dd - 1) * COL; b.style.left = x + 'px'; }
        else { x = rootX - OFF - (c._dd - 1) * COL; b.style.left = x + 'px'; b.style.transform = 'translate(-100%,-50%)'; }
        b.style.top = y + 'px';
        const w = b.offsetWidth;
        c._ax = c._dir === 1 ? { in: x, out: x + w, y } : { in: x, out: x - w, y };
        const p = n._ax; const sx = n._d === 0 ? (c._dir === 1 ? p.r : p.l) : p.out, sy = p.y;
        const ex = c._ax.in, ey = y, mx = (sx + ex) / 2;
        const path = document.createElementNS('http://www.w3.org/2000/svg', 'path');
        path.setAttribute('d', `M${sx},${sy} C${mx},${sy} ${mx},${ey} ${ex},${ey}`); path.setAttribute('class', 'mm-edge');
        svg.appendChild(path);
        placeEls(c);
      });
    };
    placeEls(root);
    made.forEach(([n, b]) => b.addEventListener('click', () => onNode(n)));
    pz.fit(true);
  }
  function drawList() {
    const li = n => {
      const has = (n.children || []).length, open = expanded.has(n.id);
      return `<li class="${has ? '' : 'leaf'}"><button data-id="${n.id}" ${has ? `aria-expanded="${open}"` : ''} aria-selected="${n.id === selId}"><span class="tw" aria-hidden="true">${has ? (open ? '−' : '+') : ''}</span><span>${esc(n.label)}</span>${n.focus ? ' <span class="tag tag-done" style="margin-left:auto">Internship focus</span>' : ''}</button>${has && open ? `<ul>${n.children.map(li).join('')}</ul>` : ''}</li>`;
    };
    list.innerHTML = `<ul class="tree" role="tree">${li(root)}</ul>`;
    $$('button[data-id]', list).forEach(b => b.addEventListener('click', () => onNode(all[b.dataset.id])));
  }
  function showPanel(n) {
    const path = []; let q = n; while (q) { path.unshift(q.label); q = q._p; }
    panel.innerHTML = n ? `<div class="fp-anim"><span class="t-label">${esc(path.join(' → '))}</span><h4 style="margin-top:6px">${esc(n.label)}</h4>${n.body ? `<p style="margin-top:4px">${n.body}</p>` : ''}${n.tags ? `<div class="g-meta">${n.tags.map(t => tag(...[].concat(t))).join('')}</div>` : ''}</div>` : '';
  }
  function onNode(n) {
    const before = new Set(Object.keys(all).filter(k => isVisible(all[k])));
    if ((n.children || []).length && n._d > 0) { expanded.has(n.id) ? collapse(n) : expanded.add(n.id); }
    selId = n.id; showPanel(n); render(before);
  }
  function collapse(n) { expanded.delete(n.id); (n.children || []).forEach(collapse); }
  function isVisible(n) { let p = n._p; while (p) { if (!expanded.has(p.id)) return false; p = p._p; } return true; }
  function render(before) {
    if (mode === 'list') drawList();
    else { const nw = before ? new Set(Object.keys(all).filter(k => isVisible(all[k]) && !before.has(k))) : new Set(); drawMap(nw); }
  }
  function setMode(m) {
    mode = m; vp.hidden = m === 'list'; list.hidden = m !== 'list';
    $$('[data-view]', el).forEach(b => b.setAttribute('aria-pressed', b.dataset.view === m));
    $$('.mm-z button', el).forEach(b => b.disabled = m === 'list');
    render();
  }
  $$('[data-view]', el).forEach(b => b.addEventListener('click', () => setMode(b.dataset.view)));
  $('[data-mm="expand"]', el).addEventListener('click', () => { Object.values(all).forEach(n => (n.children || []).length && expanded.add(n.id)); render(); });
  $('[data-mm="collapse"]', el).addEventListener('click', () => { expanded.clear(); expanded.add(root.id); render(); });
  panel.innerHTML = `<div><span class="t-label">Explore</span><p style="margin-top:8px">${cfg.intro || 'Open a branch to see how the parts connect.'}</p></div>`;
  let lastW = 0;
  new ResizeObserver(() => { const w = el.offsetWidth; if (!w || Math.abs(w - lastW) < 2) return; const first = !lastW; lastW = w; if (first) setMode(w < 700 ? 'list' : 'map'); else if (mode === 'map') pz.fit(false); }).observe(el);
}

/* =========================================================
   FLOW / STEPPER — frameworks, journeys, processes
   ========================================================= */
function wFlow(el, cfg) {
  let i = cfg.start || 0, ad = 0;
  const steps = cfg.steps;
  const adapt = cfg.adapt ? `<div class="adapt-tabs"><span class="t-label" style="margin-right:6px">${esc(cfg.adaptLabel || 'Adapt to')}</span>${cfg.adapt.map((a, k) => `<button class="chip" data-ad="${k}" aria-pressed="${k === 0}">${esc(a.name)}</button>`).join('')}</div>` : '';
  el.innerHTML = frame(cfg.title, cfg.tag ? tag(...[].concat(cfg.tag)) : '',
    `<div class="flow-rail" role="tablist" aria-label="${esc(cfg.title)}">${steps.map((s, k) => `<button class="flow-step" role="tab" data-i="${k}" aria-selected="false" tabindex="-1"><span class="dotline"><span class="dotc">${String(k + 1).padStart(2, '0')}</span><span class="ln"></span></span><span class="lab">${esc(s.t)}</span></button>`).join('')}</div>${adapt}<div class="flow-panel" role="tabpanel" aria-live="polite"></div>
     <div class="flow-nav"><button class="btn btn-sm btn-secondary" data-dir="-1">← Previous</button><span class="t-label fl-count"></span><button class="btn btn-sm btn-secondary" data-dir="1">Next →</button></div>`, cfg.hint || '');
  const panel = $('.flow-panel', el), btns = $$('.flow-step', el);
  function render() {
    btns.forEach((b, k) => { b.classList.toggle('active', k === i); b.classList.toggle('done', k < i); b.setAttribute('aria-selected', k === i); b.tabIndex = k === i ? 0 : -1; });
    const s = steps[i];
    const side = [];
    if (s.meta) Object.entries(s.meta).forEach(([k, v]) => side.push(`<div><dt>${esc(k)}</dt><dd>${v}</dd></div>`));
    if (cfg.adapt) side.push(`<div><dt>Applied to: ${esc(cfg.adapt[ad].name)}</dt><dd>${cfg.adapt[ad].notes[i]}</dd></div>`);

/* =========================================================
   PROBLEM TREE
   ========================================================= */
function wProblemTree(el, cfg) {
  const cols = cfg.cols; const items = {};
  cols.forEach((c, ci) => c.items.forEach(it => { it._c = ci; items[it.id] = it; }));
  // build adjacency
  const adj = {}; Object.keys(items).forEach(k => adj[k] = new Set());
  Object.values(items).forEach(it => (it.links || []).forEach(l => { if (items[l]) { adj[it.id].add(l); adj[l].add(it.id); } }));
  el.innerHTML = frame(cfg.title, `<button class="btn btn-sm" data-pt-clear>Clear</button>`,
    `<div class="ptree">${cols.map(c => `<div class="col"><h4>${esc(c.name)} <span>${c.items.length}</span></h4>${c.items.map(it => `<button class="pt-item" data-id="${it.id}" aria-pressed="false">${esc(it.t)}${it.tag ? `<br>${tag(...[].concat(it.tag))}` : ''}</button>`).join('')}</div>`).join('')}</div><div class="g-panel" aria-live="polite"></div>`,
    '<span>Select any item to trace its causes and effects across the tree</span><span>' + esc(cfg.note || '') + '</span>');
  const panel = $('.g-panel', el); let sel = null;
  function chain(start) {
    // traverse left (to lower column idx) and right separately so we follow a causal chain
    const res = new Set([start]);
    const walk = (k, dir) => adj[k].forEach(n => { if (!res.has(n) && Math.sign(items[n]._c - items[k]._c) === dir) { res.add(n); walk(n, dir); } });
    walk(start, -1); walk(start, 1); return res;
  }
  function paint() {
    const c = sel ? chain(sel) : null;
    $$('.pt-item', el).forEach(b => { const k = b.dataset.id; b.classList.toggle('sel', k === sel); b.classList.toggle('on', !!c && c.has(k) && k !== sel); b.classList.toggle('dim', !!c && !c.has(k)); b.setAttribute('aria-pressed', k === sel); });
    if (!sel) { panel.innerHTML = `<div><span class="t-label">How to read it</span><p style="margin-top:8px">${cfg.intro}</p></div>`; return; }
    const it = items[sel];
    const grouped = cols.map((col, ci) => ({ col, list: [...c].filter(k => items[k]._c === ci && k !== sel).map(k => items[k].t) })).filter(g => g.list.length);
    panel.innerHTML = `<div class="fp-anim"><span class="t-label">${esc(cols[it._c].name)}</span><h4 style="margin-top:6px">${esc(it.t)}</h4>${it.d ? `<p style="margin-top:4px">${it.d}</p>` : ''}<div class="grid g-3" style="margin-top:14px;gap:12px">${grouped.map(g => `<div><span class="t-label">${esc(g.col.name)}</span><ul class="list-dash t-sm" style="margin-top:6px">${g.list.map(x => `<li>${esc(x)}</li>`).join('')}</ul></div>`).join('')}</div></div>`;
  }
  $$('.pt-item', el).forEach(b => b.addEventListener('click', () => { sel = sel === b.dataset.id ? null : b.dataset.id; paint(); }));
  $('[data-pt-clear]', el).addEventListener('click', () => { sel = null; paint(); });
  paint();
}

    '<span>Drag any marker to test a different reading · Arrow keys move the selected marker</span><span>Positions are saved in this browser only.</span>');
  const mx = $('.mx', el), panel = $('.g-panel', el);
  const quad = p => (p.y >= 50 ? (p.x < 50 ? cfg.q[0] : cfg.q[1]) : (p.x < 50 ? cfg.q[2] : cfg.q[3]));
  function place() { base.forEach(b => { const p = P(b.id); const e = $(`[data-id="${b.id}"]`, mx); e.style.left = p.x + '%'; e.style.top = (100 - p.y) + '%'; e.classList.toggle('active', b.id === sel); }); }

    e.addEventListener('pointermove', ev => {
      if (!drag) return; moved = true; const r = mx.getBoundingClientRect();
      const p = pts.find(q => q.id === e.dataset.id) || (pts.push({ id: e.dataset.id, x: 50, y: 50 }), pts[pts.length - 1]);

    });
    const up = () => { if (!drag) return; drag = false; e.classList.remove('drag'); if (moved) { store.set(key, pts); info(); } };

    });
  });
  $('[data-reset]', el).addEventListener('click', () => { pts = base.map(p => ({ id: p.id, x: p.x, y: p.y })); store.set(key, pts); place(); info(); });

/* =========================================================
   CAROUSELS (scroll-snap, buttons, keyboard, mouse drag)
   ========================================================= */
function initCarousels(root) {
  $$('[data-car]', root).forEach(c => {
    const tr = $('.car-track', c), slides = $$('.car-slide', c), n = slides.length, cnt = $('.car-count', c), prog = $('.car-prog span', c);
    const prev = $('[data-prev]', c), next = $('[data-next]', c);
    const step = () => slides.length > 1 ? slides[1].offsetLeft - slides[0].offsetLeft : tr.clientWidth;
    const max = () => tr.scrollWidth - tr.clientWidth;
    const idx = () => tr.scrollLeft >= max() - 4 ? n - 1 : clamp(Math.round(tr.scrollLeft / step()), 0, n - 1);
    const go = k => tr.scrollTo({ left: clamp(k, 0, n - 1) * step(), behavior: RM.matches ? 'auto' : 'smooth' });
    let raf = 0;
    const update = () => {
      raf = 0; const k = idx(); const p = max() > 0 ? tr.scrollLeft / max() : 1;
      cnt.textContent = `${String(k + 1).padStart(2, '0')} / ${String(n).padStart(2, '0')}`;
      prog.style.transform = `scaleX(${Math.max(.04, p)})`;
      prev.disabled = tr.scrollLeft <= 4; next.disabled = tr.scrollLeft >= max() - 4;
      slides.forEach((s, j) => s.classList.toggle('is-active', j === k));
    };
    tr.addEventListener('scroll', () => { if (!raf) raf = requestAnimationFrame(update); }, { passive: true });
    prev.addEventListener('click', () => go(idx() - 1)); next.addEventListener('click', () => go(idx() + 1));
    tr.addEventListener('keydown', e => { if (e.target !== tr) return; if (e.key === 'ArrowRight') { e.preventDefault(); go(idx() + 1); } if (e.key === 'ArrowLeft') { e.preventDefault(); go(idx() - 1); } });
    let down = false, moved = false, sx = 0, sl = 0;
    tr.addEventListener('pointerdown', e => { if (e.pointerType !== 'mouse' || e.button !== 0) return; down = true; moved = false; sx = e.clientX; sl = tr.scrollLeft; });
    window.addEventListener('pointermove', e => { if (!down) return; const dx = e.clientX - sx; if (!moved && Math.abs(dx) > 6) { moved = true; tr.classList.add('dragging'); } if (moved) tr.scrollLeft = sl - dx; });
    window.addEventListener('pointerup', () => { if (!down) return; down = false; if (moved) { const v = tr.scrollLeft - sl; tr.classList.remove('dragging'); go(idx() + (Math.abs(v) > 40 && Math.round(tr.scrollLeft / step()) === Math.round(sl / step()) ? Math.sign(v) : 0)); } });
    tr.addEventListener('click', e => { if (moved) { e.preventDefault(); e.stopPropagation(); moved = false; } }, true);
    new ResizeObserver(update).observe(tr);
    update();
  });
}

const WIDGETS = { library: wLibrary, insights: wInsightWall, scanner: wScanner, graph: wGraph, mindmap: wMindmap, flow: wFlow, ptree: wProblemTree, bmc: wBMC, blueprint: wBlueprint, matrix: wMatrix, swot: wSwot, funnel: wFunnel, crm: wCRM, kpi: wKPI, timeline: wTimeline, objections: wObjections, revenue: wRevenue, capmx: wCapMatrix, lenses: wLenses };
function mountWidgets(root) {
  $$('[data-w]', root).forEach(el => {
    const fn = WIDGETS[el.dataset.w], cfg = D[el.dataset.k];
    if (!fn || !cfg) { console.warn('Missing widget', el.dataset.w, el.dataset.k); return; }
    el.classList.add('widget');
    try { fn(el, cfg); } catch (err) { console.error('Widget failed', el.dataset.k, err); }
  });
  initTabs(root);
  initCarousels(root);
  initMarquee(root);
  initLockup(root);
}

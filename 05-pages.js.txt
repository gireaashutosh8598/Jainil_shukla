function initContact(root) {
  $$('[data-openbot]', root).forEach(b => b.addEventListener('click', () => { const l = document.querySelector('.jv-launch'); if (l && l.getAttribute('aria-expanded') !== 'true') l.click(); }));
  const f = $('#contactForm', root); if (!f) return;
  const msg = f.elements.message, cc = $('#charCount', root);
  const count = () => { cc.textContent = `${msg.value.length} / 2000`; };
  msg.addEventListener('input', count);
  $$('[data-int]', root).forEach(b => b.addEventListener('click', () => {
    const it = INTENTS[+b.dataset.int];
    $$('[data-int]', root).forEach(x => x.setAttribute('aria-pressed', x === b));
    f.elements.stage.value = it[1];
    if (!msg.value.trim() || INTENTS.some(x => msg.value === x[3])) msg.value = it[3];
    count();
    const t = $('#ct-form', root); window.scrollTo({ top: t.getBoundingClientRect().top + scrollY - 90, behavior: RM.matches ? 'auto' : 'smooth' });
    setTimeout(() => { const n = f.elements.name; (n.value ? msg : n).focus({ preventScroll: true }); }, RM.matches ? 0 : 600);
  }));
  $('#formMode', root).textContent = FORM_ENDPOINT ? 'Sends directly to Jainil.' : 'Opens your email app with this message ready to send.';
  $('#copyEmail', root).addEventListener('click', async () => {
    try { await navigator.clipboard.writeText(EMAIL); $('#copyMsg', root).textContent = 'Copied'; } catch (e) { $('#copyMsg', root).textContent = EMAIL; }
    setTimeout(() => $('#copyMsg', root).textContent = '', 2500);
  });
  f.addEventListener('submit', async e => {
    e.preventDefault();
    const v = k => f.elements[k].value.trim(); let ok = true;
    const set = (k, msg) => { const i = f.elements[k]; i.setAttribute('aria-invalid', !!msg); $('#e-' + (k === 'message' ? 'msg' : k), root).textContent = msg || ''; if (msg && ok) { i.focus(); ok = false; } };
    set('name', v('name') ? '' : 'Please add your name.');
    set('email', /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(v('email')) ? '' : 'Please add a valid email address.');
    set('message', v('message').length > 4 ? '' : 'Please add a short message.');
    if (!ok) return;
    const st = $('#formStatus', root);
    const body = `Name: ${v('name')}\nEmail: ${v('email')}\nOrganisation: ${v('organisation') || '-'}\nStage: ${v('stage')}\n\n${v('message')}`;
    if (FORM_ENDPOINT) {
      st.className = 'form-status show ok'; st.textContent = 'Sending…';
      try {
        const r = await fetch(FORM_ENDPOINT, { method: 'POST', headers: { 'Content-Type': 'application/json', Accept: 'application/json' }, body: JSON.stringify({ name: v('name'), email: v('email'), organisation: v('organisation'), stage: v('stage'), message: v('message') }) });
        if (!r.ok) throw new Error(r.status);
        st.textContent = 'Thank you. Your message has been sent.'; f.reset();
      } catch (err) { st.className = 'form-status show bad'; st.innerHTML = `Something went wrong sending the form. Please email <a class="link" href="mailto:${EMAIL}">${EMAIL}</a> directly.`; }
    } else {
      window.location.href = `mailto:${EMAIL}?subject=${encodeURIComponent('Let\'s talk: ' + v('stage'))}&body=${encodeURIComponent(body)}`;
      st.className = 'form-status show ok'; st.innerHTML = `Your email app should now open with the message ready. If nothing happened, email <a class="link" href="mailto:${EMAIL}">${EMAIL}</a> directly.`;
    }
  });
}

function initReveal(root) {
  if (rvObs) rvObs.disconnect();
  const els = $$('.rv', root);
  if (RM.matches || !('IntersectionObserver' in window)) { els.forEach(e => e.classList.add('in')); return; }
  rvObs = new IntersectionObserver(es => es.forEach(e => { if (e.isIntersecting) { e.target.classList.add('in'); rvObs.unobserve(e.target); } }), { rootMargin: '0px 0px -8% 0px', threshold: .05 });
  els.forEach(e => rvObs.observe(e));
}
function initSpy(root) {
  if (spyObs) spyObs.disconnect();
  const btns = $$('[data-scrollto]', root); if (!btns.length) return;
  btns.forEach(b => b.addEventListener('click', () => {
    const t = document.getElementById(b.dataset.scrollto); if (!t) return;
    const y = t.getBoundingClientRect().top + window.scrollY - (window.innerWidth <= 1060 ? 130 : 90);
    window.scrollTo({ top: y, behavior: RM.matches ? 'auto' : 'smooth' });
    const h = $('h2', t); if (h) { h.setAttribute('tabindex', '-1'); setTimeout(() => h.focus({ preventScroll: true }), RM.matches ? 0 : 500); }
  }));
  const map = new Map(btns.map(b => [b.dataset.scrollto, b]));

'auto' : 'smooth' }); } } });
  }, { rootMargin: '-30% 0px -60% 0px' });
  $$('.cs-sec[id]', root).forEach(s => spyObs.observe(s));
}

function helpHTML(title, cards, vs, fb, labels = ['What I did.', 'How it helped.'], right = 'My approach') {
  return `<div class="help-grid">${cards.map(([t, did, why], i) => `<article class="help-c rv"><span class="help-n">${String(i + 1).padStart(2, '0')}</span><h4 class="t-h4">${esc(t)}</h4><p class="t-sm"><b>${labels[0]}</b> ${esc(did)}</p><p class="t-sm ink2"><b>${labels[1]}</b> ${esc(why)}</p></article>`).join('')}</div>
  <div class="vs rv"><div class="vs-h"><span class="t-label">${esc(title)}</span><span></span><span class="t-label" style="color:var(--blue-ink)">${esc(right)}</span></div>${vs.map(([a, b]) => `<div class="vs-r"><span>${esc(a)}</span><span class="vs-a" aria-hidden="true">→</span><strong>${esc(b)}</strong></div>`).join('')}</div>
  ${fb ? `<p class="note" style="margin-top:14px"><strong>${esc(fb)}</strong> ${ph('ADD A LINE OF FEEDBACK, IF APPROVED')}</p>` : ''}`;
}

/* ---------- value chain widget ---------- */
function wChain(el, cfg) {
  let i = 0; const L = cfg.labels || ['Struggle', 'CLIQUE answer', 'Pillar'];
  el.innerHTML = frame(cfg.title, tag(...[].concat(cfg.tag)), `<div class="ch-rail">${cfg.steps.map((s, k) => `<button class="ch-s" data-k="${k}" aria-pressed="${k === 0}"><span class="ch-n">${String(k + 1).padStart(2, '0')}</span><span class="ch-t">${esc(s.t)}</span><span class="ch-m" aria-hidden="true">${'●'.repeat(s.p)}<i>${'●'.repeat(3 - s.p)}</i></span></button>`).join('')}</div><div class="g-panel" aria-live="polite"></div>`, `<span>${esc(cfg.hint || 'Dots show how hard each step is for an independent creator to handle alone (framing, to be validated)')}</span>`);
  const panel = $('.g-panel', el);
  const r = () => { const s = cfg.steps[i]; $$('.ch-s', el).forEach((b, k) => b.setAttribute('aria-pressed', k === i)); panel.innerHTML = `<div class="fp-anim"><span class="t-label">Step ${i + 1} of ${cfg.steps.length}</span><h4 style="margin-top:6px">${esc(s.t)}</h4><dl class="kv" style="margin-top:10px"><dt>${L[0]}</dt><dd>${esc(s.pain)}</dd><dt>${L[1]}</dt><dd>${esc(s.fix)}</dd><dt>${L[2]}</dt><dd>${esc(s.pillar)}</dd></dl></div>`; };
  $$('.ch-s', el).forEach(b => b.addEventListener('click', () => { i = +b.dataset.k; r(); }));
  r();
}
WIDGETS.chain = wChain;

/* ---------- value proposition canvas widget ---------- */
function wVPC(el, cfg) {
  let side = 0, sel = null;
  el.innerHTML = frame(cfg.title, tag('hyp', 'To be validated'), `<div class="vpc-tabs chips" role="group" aria-label="Customer side" style="padding:12px 18px">${cfg.sides.map((s, k) => `<button class="chip" data-side="${k}" aria-pressed="${k === 0}">${esc(s.name)}</button>`).join('')}</div><div class="vpc"></div><div class="g-panel" aria-live="polite"></div>`, '<span>Select any job, pain or gain to see what answers it</span>');
  const box = $('.vpc', el), panel = $('.g-panel', el);
  function draw() {
    const S = cfg.sides[side];
    const list = (arr, cls) => arr.map((x, k) => `<button class="vpc-i ${cls}" data-id="${cls}${k}" aria-pressed="false">${esc(x[0])}</button>`).join('');
    box.innerHTML = `<div class="vpc-map"><span class="t-label">What CLIQUE offers</span><div class="vpc-sq"><div><b>Products & services</b>${S.prod.map(p => `<span class="vpc-p">${esc(p)}</span>`).join('')}</div><div><b>Pain relievers</b>${list(S.pains.map(p => [p[1]]), 'r')}</div><div><b>Gain creators</b>${list(S.gains.map(g => [g[1]]), 'c')}</div></div></div>
      <div class="vpc-arrow" aria-hidden="true">⇄</div>
      <div class="vpc-prof"><span class="t-label">${esc(S.name)}</span><div class="vpc-ci"><div><b>Jobs</b>${S.jobs.map(j => `<span class="vpc-p">${esc(j)}</span>`).join('')}</div><div><b>Pains</b>${list(S.pains, 'p')}</div><div><b>Gains</b>${list(S.gains, 'g')}</div></div></div>`;
    $$('.vpc-i', box).forEach(b => b.addEventListener('click', () => { const id = b.dataset.id; sel = sel === id ? null : id; paint(); }));
    paint();
  }
  function paint() {
    const S = cfg.sides[side];
    const pair = sel ? ({ p: 'r', r: 'p', g: 'c', c: 'g' })[sel[0]] + sel.slice(1) : null;
    $$('.vpc-i', box).forEach(b => { b.setAttribute('aria-pressed', b.dataset.id === sel); b.classList.toggle('pair', b.dataset.id === pair); b.classList.toggle('dim', !!sel && b.dataset.id !== sel && b.dataset.id !== pair); });
    if (!sel) { panel.innerHTML = `<div><span class="t-label">Reading the canvas</span><p style="margin-top:8px">${esc(S.intro)}</p></div>`; return; }
    const k = +sel.slice(1), isPain = sel[0] === 'p' || sel[0] === 'r', it = (isPain ? S.pains : S.gains)[k];
    panel.innerHTML = `<div class="fp-anim"><span class="t-label">${isPain ? 'Pain → reliever' : 'Gain → creator'}</span><h4 style="margin-top:6px">${esc(it[0])}</h4><p style="margin-top:4px">→ ${esc(it[1])}</p></div>`;
  }
  $$('[data-side]', el).forEach(b => b.addEventListener('click', () => { side = +b.dataset.side; sel = null; $$('[data-side]', el).forEach(x => x.setAttribute('aria-pressed', x === b)); draw(); }));
  draw();
}
WIDGETS.vpc = wVPC;

function wStoryboard(el, cfg) {
  const F = cfg.frames; let i = 0, timer = null;
  const svg = k => `<svg viewBox="0 0 320 180" class="sb-svg" aria-hidden="true"><rect width="320" height="180" fill="var(--bg-2)"/>${SCENES[k]}</svg>`;
  el.innerHTML = `<div class="sb">
    <div class="sb-stage"><div class="sb-frame">${svg(0)}<span class="sb-n"></span></div><div class="sb-text" aria-live="polite"></div></div>
    <div class="sb-ctl"><button class="btn btn-sm btn-secondary" data-sb="-1" aria-label="Previous scene">←</button><button class="btn btn-sm btn-primary sb-play" type="button" aria-pressed="false">Play the story</button><button class="btn btn-sm btn-secondary" data-sb="1" aria-label="Next scene">→</button><span class="t-label sb-c"></span></div>
    <div class="sb-strip" role="tablist" aria-label="Storyboard frames">${F.map((f, k) => `<button role="tab" class="sb-thumb" data-k="${k}" aria-selected="${k === 0}" aria-label="Scene ${k + 1}: ${esc(f.when)}, ${esc(f.t)}"><span class="sb-tf">${svg(k)}</span><span class="sb-tl"><b>${String(k + 1).padStart(2, '0')}</b> ${esc(f.when)}</span></button>`).join('')}</div>
  </div>`;
  const frame = $('.sb-frame', el), text = $('.sb-text', el), play = $('.sb-play', el);
  function go(k, user) {
    i = (k + F.length) % F.length; const f = F[i];
    frame.innerHTML = `${svg(i)}<span class="sb-n">Scene ${String(i + 1).padStart(2, '0')} / ${String(F.length).padStart(2, '0')}</span>`;
    frame.classList.remove('sb-in'); void frame.offsetWidth; frame.classList.add('sb-in');
    text.innerHTML = `<div class="fp-anim"><span class="t-label">${esc(f.when)} ${tag(f.tag)}</span><h3 class="t-h3">${esc(f.t)}</h3><p class="sb-scene">${esc(f.scene)}</p><p>${f.caption}</p><p class="sb-thread"><b>Thread to today</b>${esc(f.thread)}</p></div>`;
    $$('.sb-thumb', el).forEach((b, j) => b.setAttribute('aria-selected', j === i));
    $('.sb-c', el).textContent = `${i + 1} / ${F.length}`;
    if (user) stop();
  }
  function stop() { clearInterval(timer); timer = null; play.textContent = 'Play the story'; play.setAttribute('aria-pressed', 'false'); }
  play.addEventListener('click', () => { if (timer) return stop(); play.textContent = 'Pause'; play.setAttribute('aria-pressed', 'true'); if (i === F.length - 1) go(0); timer = setInterval(() => { if (!el.isConnected || el.closest('[hidden]') || i === F.length - 1) { stop(); return; } go(i + 1); }, RM.matches ? 4000 : 3200); });
  $$('[data-sb]', el).forEach(b => b.addEventListener('click', () => go(i + +b.dataset.sb, true)));
  $$('.sb-thumb', el).forEach(b => b.addEventListener('click', () => go(+b.dataset.k, true)));
  $('.sb-strip', el).addEventListener('keydown', e => { if (e.key === 'ArrowRight' || e.key === 'ArrowLeft') { e.preventDefault(); go(i + (e.key === 'ArrowRight' ? 1 : -1), true); $$('.sb-thumb', el)[i].focus(); } });
  go(cfg.start || 0);
}

function wAnatomy(el, cfg) {
  let v = 0, s = 0;
  el.innerHTML = frame('Anatomy of the dispenser · explored vs proposed', `<div class="chips" role="group" aria-label="View">${cfg.views.map((x, k) => `<button class="chip" data-v="${k}" aria-pressed="${k === 0}">${esc(x.name)}</button>`).join('')}</div>`,
    `<div class="an"><div class="an-fig"><svg viewBox="0 0 300 360" aria-hidden="true"></svg><div class="an-spots"></div></div><div class="an-out" aria-live="polite"></div></div>`,
    '<span>Select a numbered part to see what it does and where problems can start</span><span>General mechanism understanding, simplified. Not an engineering drawing.</span>');
  const svg = $('svg', el), spots = $('.an-spots', el), out = $('.an-out', el);
  function draw() {
    const V = cfg.views[v]; svg.innerHTML = V.svg;
    spots.innerHTML = V.spots.map(([x, y, t], k) => `<button class="an-dot" style="left:${x}%;top:${y}%" data-s="${k}" aria-pressed="${k === s}" aria-label="${k + 1}. ${esc(t)}">${k + 1}</button>`).join('');
    $$('.an-dot', el).forEach(b => b.addEventListener('click', () => { s = +b.dataset.s; info(); }));
    $$('[data-v]', el).forEach(b => b.setAttribute('aria-pressed', +b.dataset.v === v));
    info();
  }
  function info() {
    const V = cfg.views[v], [, , t, d] = V.spots[s];
    $$('.an-dot', el).forEach(b => b.setAttribute('aria-pressed', +b.dataset.s === s));
    out.innerHTML = `<div class="fp-anim"><div style="margin-bottom:10px">${tag(...V.tag)}</div><span class="t-label">Part ${s + 1} of ${V.spots.length}</span><h4 class="t-h3" style="margin:6px 0 8px">${esc(t)}</h4><p>${esc(d)}</p><ol class="an-list">${V.spots.map(([, , n], k) => `<li class="${k === s ? 'on' : ''}"><button data-li="${k}">${k + 1}. ${esc(n)}</button></li>`).join('')}</ol></div>`;
    $$('[data-li]', out).forEach(b => b.addEventListener('click', () => { s = +b.dataset.li; info(); }));
  }
  $$('[data-v]', el).forEach(b => b.addEventListener('click', () => { v = +b.dataset.v; s = 0; draw(); }));
  draw();
}

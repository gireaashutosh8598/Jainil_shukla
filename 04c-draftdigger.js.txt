/* ---------- widgets ---------- */
function wSituations(el, cfg) {
  let i = 0;
  el.innerHTML = `<div class="sit"><div class="sit-tabs" role="tablist" aria-label="Your situation">${cfg.items.map((x, k) => `<button role="tab" aria-selected="${k === 0}" tabindex="${k === 0 ? 0 : -1}" data-k="${k}"><span class="t-label">0${k + 1}</span><strong>${esc(x.t)}</strong></button>`).join('')}</div><div class="sit-out" role="tabpanel" aria-live="polite"></div></div>`;
  const out = $('.sit-out', el), bs = $$('.sit-tabs button', el);
  const STAGES = ['Understand', 'Diagnose', 'Strategise', 'Structure', 'Coordinate', 'Execute'];
  const r = () => {
    const x = cfg.items[i]; bs.forEach((b, k) => { b.setAttribute('aria-selected', k === i); b.tabIndex = k === i ? 0 : -1; });
    out.innerHTML = `<div class="fp-anim"><p class="sit-d">${esc(x.d)}</p><div class="grid g-3" style="gap:16px;margin-top:18px">
      <div><span class="t-label">Questions we'd answer</span><ul class="list-dash t-sm" style="margin-top:8px">${x.q.map(q => `<li>${esc(q)}</li>`).join('')}</ul></div>
      <div><span class="t-label">Services that apply</span><div class="chips" style="margin-top:8px">${x.svc.map(s => `<span class="pill">${esc(s)}</span>`).join('')}</div></div>
      <div><span class="t-label">Where the process focuses</span><div class="sit-st">${STAGES.map(s => `<span class="${x.st.includes(s) ? 'on' : ''}">${s}</span>`).join('')}</div></div></div></div>`;
  };
  bs.forEach(b => b.addEventListener('click', () => { i = +b.dataset.k; r(); }));
  $('.sit-tabs', el).addEventListener('keydown', e => { if (e.key === 'ArrowRight' || e.key === 'ArrowLeft') { e.preventDefault(); i = (i + (e.key === 'ArrowRight' ? 1 : -1) + bs.length) % bs.length; r(); bs[i].focus(); } });
  r();
}
function wServices(el, cfg) {
  const all = cfg.groups.flatMap(g => g.items.map(it => ({ g: g.g, c: g.c, it })));
  let sel = 0;
  el.innerHTML = frame('Envisioned services · select one to see the detail', tag('concept', 'Envisioned'),
    `<div class="svc"><div class="svc-groups">${cfg.groups.map(g => `<div class="svc-g"><span class="t-label svc-gl svc-${g.c}">${esc(g.g)}</span>${g.items.map(it => { const k = all.findIndex(a => a.it === it); return `<button class="svc-b" data-k="${k}" aria-pressed="${k === 0}">${esc(it[0])}</button>`; }).join('')}</div>`).join('')}</div><div class="svc-out" aria-live="polite"></div></div>`,
    '<span>Each service maps to a stage of the process and to work Jainil has already done</span>');
  const out = $('.svc-out', el);
  const r = () => {
    const { g, it } = all[sel]; const [n, q, o, st, href, proof] = it;
    $$('.svc-b', el).forEach(b => b.setAttribute('aria-pressed', +b.dataset.k === sel));
    out.innerHTML = `<div class="fp-anim"><span class="t-label">${esc(g)}</span><h3 class="t-h3" style="margin:8px 0 14px">${esc(n)}</h3><dl class="kv"><dt>Answers</dt><dd>${esc(q)}</dd><dt>You get</dt><dd>${esc(o)}</dd><dt>Stage</dt><dd>${esc(st)}</dd><dt>Built on</dt><dd><a class="link" href="${href}">${esc(proof)} →</a></dd></dl></div>`;
  };
  $$('.svc-b', el).forEach(b => b.addEventListener('click', () => { sel = +b.dataset.k; r(); }));
  r();
}
